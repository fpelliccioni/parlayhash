cmake_minimum_required(VERSION 3.25)
project(parlay_concurrency_tests LANGUAGES CXX)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 23)
endif()
enable_testing()
find_package(Threads REQUIRED)
set(SANITIZER "none" CACHE STRING "none, address, undefined, thread, leak, or memory")
set_property(CACHE SANITIZER PROPERTY STRINGS none address undefined thread leak memory)
set(MSAN_LIBCXX_ROOT "" CACHE PATH "MemorySanitizer-instrumented libc++ installation")
if(NOT SANITIZER MATCHES "^(none|address|undefined|thread|leak|memory)$")
  message(FATAL_ERROR "Unknown SANITIZER=${SANITIZER}")
endif()
if(SANITIZER STREQUAL "memory")
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MSAN_LIBCXX_ROOT)
    message(FATAL_ERROR "MSan requires Clang and -DMSAN_LIBCXX_ROOT pointing to an instrumented libc++")
  endif()
endif()
add_executable(concurrency_stress stress.cpp)
target_compile_features(concurrency_stress PRIVATE cxx_std_23)
set_target_properties(concurrency_stress PROPERTIES CXX_STANDARD ${CMAKE_CXX_STANDARD} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_include_directories(concurrency_stress PRIVATE ../../include)
target_link_libraries(concurrency_stress PRIVATE Threads::Threads)
target_compile_options(concurrency_stress PRIVATE -O1 -g -fno-omit-frame-pointer)
if(NOT SANITIZER STREQUAL "none")
  target_compile_options(concurrency_stress PRIVATE -fsanitize=${SANITIZER} -fno-sanitize-recover=all)
  target_link_options(concurrency_stress PRIVATE -fsanitize=${SANITIZER})
endif()
if(SANITIZER STREQUAL "memory")
  target_compile_options(concurrency_stress PRIVATE -fsanitize-memory-track-origins=2
    -stdlib=libc++ -nostdinc++ -isystem "${MSAN_LIBCXX_ROOT}/include/c++/v1" -fPIE)
  target_link_options(concurrency_stress PRIVATE -stdlib=libc++ -pie
    "-L${MSAN_LIBCXX_ROOT}/lib" "-Wl,-rpath,${MSAN_LIBCXX_ROOT}/lib")
endif()
foreach(scenario snapshot_api atomic_snapshot atomic_padded_snapshot padded_churn direct_churn indirect_churn direct_growth indirect_growth upsert string_upsert string_overflow_upsert)
  add_test(NAME ${scenario} COMMAND concurrency_stress ${scenario})
  set_tests_properties(${scenario} PROPERTIES TIMEOUT 120
    ENVIRONMENT "PARLAY_NUM_THREADS=4;ASAN_OPTIONS=detect_leaks=1:halt_on_error=1;TSAN_OPTIONS=halt_on_error=1")
endforeach()
